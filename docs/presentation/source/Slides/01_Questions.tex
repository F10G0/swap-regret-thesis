\begin{frame}
  \frametitle{This thesis asks three questions}

  \vspace{0.6cm}
  \begin{enumerate}
    \setlength{\itemsep}{0.48cm}

    \uncover<1->{%
    \item \textbf{Regret notions.}\quad
      How do algorithms differ under external, internal, and swap regret
      at finite horizons?
    }

    \uncover<2->{%
    \item \textbf{Robustness.}\quad
      How stable are these differences across feedback models,
      environments, action spaces, and horizons?
    }

    \uncover<3->{%
    \item \textbf{Repeated games.}\quad
      How do regret differences appear in CE / CCE behavior,
      and how strongly do they depend on the opponent?
    }

  \end{enumerate}
\end{frame}
